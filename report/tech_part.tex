\chapter{Технологическая часть}

Для программной реализации серверной части приложения, которая осуществляет запросы к базе данных был выбран язык TypeScript~\cite{tsdoc} с библиотекой Express~\cite{expressdoc}, предназначенной для создания веб-серверов и обработки HTTP-запросов. Для реализации пользовательской части использовался язык разметки HTML~\cite{htmldoc}, язык таблиц стилей CSS~\cite{cssdoc} и для реализации взаимодействия пользователя с интерфейсом использовался язык Typescript~\cite{tsdoc}. Для сборки и запуска пользовательской части приложения использовался инструмент Vite~\cite{vitedoc}.  
В качестве системы управления базами данных использовалась PostgreSQL~\cite{postgresqldoc}, обеспечивающая хранение и обработку данных приложения.  
Для реализации механизма кеширования и хранения временных данных использовалась система Redis~\cite{redisdoc}.  
Разработкаосуществлялась в программе VScode~\cite{vscode}.

\subsection{Создание триггеров и хранимой процедуры}

На листинге~\ref{proc_register_team} представлен код программы для создания хранимой процедуры регистрации команды на турнир.

\begin{lstlisting}[language=SQL, caption=\raggedright{Создание процедуры регистрации команды}, label=proc_register_team]
	create or replace procedure sport.register_team(p_team_id bigint, p_tournament_id bigint)
	language plpgsql
	as $$
	declare
	v_status text;
	v_exists boolean;
	begin
	select status into v_status
	from sport.tournament
	where id = p_tournament_id;
	
	if not found then
	raise exception 'Турнир % не найден', p_tournament_id;
	end if;
	
	if v_status != 'registration' then
	raise exception 'Регистрация на турнир закрыта (текущий статус: %)', v_status;
	end if;
	
	select exists(
	select 1 from sport.tournament_registration 
	where tournament_id = p_tournament_id and team_id = p_team_id
	) into v_exists;
	
	if v_exists then
	raise exception 'Команда % уже подала заявку на турнир %', p_team_id, p_tournament_id;
	end if;
	
	insert into sport.tournament_registration(tournament_id, team_id, registration_date, status_registration)
	values (p_tournament_id, p_team_id, current_timestamp, 'pending');
	end;
	$$;
\end{lstlisting}

На листинге~\ref{trg_generate_schedule} представлен код программы для создания триггера и триггерной функции автоматической генерации расписания турнира.

\begin{lstlisting}[language=SQL, caption=\raggedright{Создание триггера генерации расписания турнира}, label=trg_generate_schedule]
	create or replace function sport.trg_generate_tournament_schedule()
	returns trigger
	language plpgsql
	as $$
	declare
	v_team_ids bigint[];
	v_loc_ids bigint[];
	v_team_count int;
	v_loc_count int;
	v_match_id bigint;
	v_loc_idx int;
	v_date timestamp;
	i int;
	j int;
	begin
	if old.status = 'registration' and new.status = 'active' then
	
	select array_agg(team_id) into v_team_ids
	from sport.tournament_registration
	where tournament_id = new.id and status_registration = 'approved';
	
	select array_agg(id) into v_loc_ids
	from sport.location;
	
	v_team_count := coalesce(array_length(v_team_ids, 1), 0);
	v_loc_count := coalesce(array_length(v_loc_ids, 1), 0);
	
	if v_team_count >= 2 then
	v_date := coalesce(new.start_date::timestamp, current_timestamp);
	
	for i in 1 .. (v_team_count - 1) loop
	for j in (i + 1) .. v_team_count loop
	
	v_loc_idx := floor(random() * v_loc_count + 1)::int;
	
	insert into sport.match (tournament_id, location_id, match_date, stage_name, status)
	values (
	new.id, 
	v_loc_ids[v_loc_idx], 
	v_date,
	'Групповой этап', 
	'scheduled'
	) returning id into v_match_id;
	
	insert into sport.match_participant (match_id, team_id, score)
	values (v_match_id, v_team_ids[i], null);
	
	insert into sport.match_participant (match_id, team_id, score)
	values (v_match_id, v_team_ids[j], null);
	
	v_date := v_date + interval '1 day';
	
	end loop;
	end loop;
	end if;
	end if;
	
	return new;
	end;
	$$;
	
	drop trigger if exists trg_generate_schedule on sport.tournament;
	
	create trigger trg_generate_schedule
	after update of status on sport.tournament
	for each row
	execute function sport.trg_generate_tournament_schedule();
\end{lstlisting}

На листинге~\ref{trg_finish_tournament} представлен код программы для создания триггера и триггерной функции автоматического завершения турнира.

\begin{lstlisting}[language=SQL, caption=\raggedright{Создание триггера завершения турнира}, label=trg_finish_tournament]
	create or replace function sport.trg_finish_tournament()
	returns trigger
	language plpgsql
	as $$
	begin
	if new.status = 'finished' and old.status is distinct from 'finished' and new.stage_name = 'Финал' then
	
	update sport.tournament 
	set status = 'completed' 
	where id = new.tournament_id;
	
	end if;
	
	return new;
	end;
	$$;
	
	drop trigger if exists trg_match_finish_tournament on sport.match;

\section{Интерфейс программной реалиазции}

\subsection{Интерфейс взаимодействия с базой данных}
Для работы с базой данных был создан API при помощи библиотеки Express~\cite{expressdoc}. В программном интерфейсе реализованы методы для выполнения операций создания, чтения и удаления всех созданных сущностей в базе данных.

Методы взаимодействия с таблицей турниров базы данных:
\begin{itemize}
	\item получение всех турниров;
	\item получение турнира по идентификатору;
	\item получение всех турниров по идентификатору организатора;
	\item создание турнира;
	\item обновление информации о турнире.
\end{itemize}

Методы взаимодействия с таблицей матчей базы данных:
\begin{itemize}
	\item получение всех матчей с возможностью фильтрации;
	\item получение матча по идентификатору;
	\item получение всех матчей по идентификатору турнира;
	\item получение истории матчей между двумя заданными командами;
	\item получение списка команд с одобренными заявками для участия в турнире;
	\item создание матча;
	\item обновление информации о матче.
\end{itemize}

Методы взаимодействия с таблицей заявок на турниры базы данных:
\begin{itemize}
	\item получение всех турниров, доступных для регистрации конкретной команды;
	\item получение всех заявок по идентификатору команды;
	\item получение всех заявок по идентификатору организатора;
	\item создание заявки на участие в турнире;
	\item обновление статуса заявки -- подтверждение или отклонение.
\end{itemize}

Методы взаимодействия с таблицей команд базы данных:
\begin{itemize}
	\item получение полной информации о команде по идентификатору;
	\item получение идентификатора команды по названию;
	\item получение названия команды по идентификатору;
	\item получение команды по идентификатору пользователя-менеджера.
\end{itemize}

Методы взаимодействия с таблицей игроков базы данных:
\begin{itemize}
	\item получение состава команды по идентификатору;
	\item добавление нового игрока в команду;
	\item удаление игрока из команды.
\end{itemize}

Методы взаимодействия с таблицей организаторов базы данных:
\begin{itemize}
	\item получение организатора по идентификатору;
	\item получение организатора по идентификатору пользователя.
\end{itemize}

Методы взаимодействия с таблицей пользователей базы данных:
\begin{itemize}
	\item регистрация нового пользователя;
	\item авторизация пользователя.
\end{itemize}
